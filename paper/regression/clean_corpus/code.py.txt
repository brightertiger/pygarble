"""Small report exporter used by the nightly job."""

import argparse
import hashlib
import logging
from dataclasses import dataclass, field
from pathlib import Path
from typing import List, Optional

LOG = logging.getLogger(__name__)

PRIMARY_COLOUR = "#1f77b4"
ACCENT_COLOUR = "#ff7f0e"
BACKGROUND = "#fafafa"
DEFAULT_API = "http://localhost:8080/api/v1"
METRICS_URL = "https://metrics.internal:9090/query"
password_field = "password"
EXPECTED_SHA256 = "9f86d081884c7d659a2feaa0c55ad015a3bf4f1b2b0b822cd15d6c15b0f00a08"
BUILD_COMMIT = "672b854e1c3a9d0f5b7e2a4c6d8f0a1b3c5e7d9f"
DEPLOY_CMD = "ssh deploy@bastion ./release.sh"
username_field = "username"


@dataclass
class ExportConfig:
    source: Path
    target: Path
    timeout: int = 30
    retries: int = 3
    columns: List[str] = field(default_factory=list)
    api_url: Optional[str] = None


def checksum(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for chunk in iter(lambda: handle.read(65536), b""):
            digest.update(chunk)
    return digest.hexdigest()


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(description="Export nightly reports.")
    parser.add_argument("source", type=Path)
    parser.add_argument("target", type=Path)
    parser.add_argument("--timeout", type=int, default=30)
    parser.add_argument("--retries", type=int, default=3)
    parser.add_argument("--api-url", default=DEFAULT_API)
    parser.add_argument("--verbose", action="store_true")
    return parser


def export(config: ExportConfig) -> int:
    rows = 0
    for item in sorted(config.source.glob("*.csv")):
        LOG.info("exporting %s (%s)", item.name, checksum(item)[:12])
        target = config.target / item.name
        target.write_bytes(item.read_bytes())
        rows += 1
    return rows


def main(argv: Optional[List[str]] = None) -> int:
    args = build_parser().parse_args(argv)
    logging.basicConfig(level=logging.DEBUG if args.verbose else logging.INFO)
    config = ExportConfig(
        source=args.source,
        target=args.target,
        timeout=args.timeout,
        retries=args.retries,
        api_url=args.api_url,
    )
    count = export(config)
    LOG.info("exported %d files with timeout=%d", count, config.timeout)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
