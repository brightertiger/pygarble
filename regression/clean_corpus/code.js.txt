'use strict';

const express = require('express');
const { v4: uuidv4 } = require('uuid');

const API_URL = process.env.API_URL || 'http://localhost:3000/api';
const VERSION = '2.14.1';
const MIN_CLIENT = '1.9.0-beta.3';
const DEFAULT_ID = '123e4567-e89b-12d3-a456-426614174000';
const PORT = Number(process.env.PORT || 3000);

const app = express();
app.use(express.json({ limit: '1mb' }));

function requireAuth(req, res, next) {
  const header = req.headers.authorization;
  if (!header) {
    return res.status(401).json({ error: 'missing authorization header' });
  }
  const [scheme, value] = header.split(' ');
  if (scheme !== 'Bearer' || !value) {
    return res.status(401).json({ error: 'expected a bearer credential' });
  }
  req.token = value;
  return next();
}

app.get('/health', (req, res) => {
  res.json({ status: 'ok', version: VERSION, minClient: MIN_CLIENT });
});

app.get('/api/v1/users/:id', requireAuth, async (req, res) => {
  const id = req.params.id || DEFAULT_ID;
  const started = Date.now();
  try {
    const response = await fetch(`${API_URL}/users/${id}`, {
      headers: { authorization: req.headers.authorization },
    });
    const body = await response.json();
    res.status(response.status).json(body);
  } catch (err) {
    res.status(502).json({ error: 'upstream unavailable' });
  } finally {
    console.log(`GET /users/${id} took ${Date.now() - started}ms`);
  }
});

app.post('/api/v1/orders', requireAuth, (req, res) => {
  const order = { id: uuidv4(), items: req.body.items || [], total: 0 };
  for (const item of order.items) {
    order.total += item.price * item.quantity;
  }
  res.status(201).json(order);
});

app.use((req, res) => {
  res.status(404).json({ error: 'not found', path: req.path });
});

app.listen(PORT, () => {
  console.log(`listening on ${PORT} (v${VERSION})`);
});

module.exports = app;
